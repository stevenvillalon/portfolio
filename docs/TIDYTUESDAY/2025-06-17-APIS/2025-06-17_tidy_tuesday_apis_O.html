Question of Interest

In a single chart, demonstrate the ratio of nominations that came from each party, the gender of nominees, and how long they served.

Goal: Make a Sankey diagram showing the flow of nominations through gender and length of service.

5. Build App

# Filter 1: Provider Name
selectInput("provider", "Provider Name",
            choices = sort(unique(df$provider_name)),
            selected = NULL)
renderUI({
  req(input$provider)

  filtered_services <- df %>%
    filter(provider_name == input$provider) %>%
    pull(service_name) %>%
    unique() %>%
    sort()

  selectInput("service", "Service Name",
              choices = filtered_services,
              selected = NULL)
})
renderUI({
  req(input$provider, input$service)

  row <- df %>%
    filter(provider_name == input$provider,
           service_name == input$service) %>%
    slice(1)

  tagList(
    tags$img(src = row$url, height = "100px", style = "margin-bottom:1rem;"),
    tags$h3(row$provider_name),
    tags$p(tags$strong("Service Name: "), row$service_name),
    tags$p(tags$strong("Description: "), row$description),
    tags$p(tags$strong("Category: "), row$apisguru_category),
    tags$p(tags$strong("Date Added: "), row$added),
    tags$p(tags$strong("Latest Update: "), row$updated)
    )
})
